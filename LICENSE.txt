Here is a complete, production-structured React + TypeScript + Tailwind CSS
frontend implementation for CodzSkill Labs, designed to match the visual layout,
color palette, typography, and dashboard workflows of your reference screenshot
while remaining cleanly architected for integration with an ASP.NET Core 10 Web
API backend.

export function formatInr(amount: number): string {
  return new Intl.NumberFormat('en-IN', {
    style: 'currency',
    currency: 'INR',
    maximumFractionDigits: 0,
  }).format(amount);
}

export function cn(...classes: Array<string | false | null | undefined>): string {
  return classes.filter(Boolean).join(' ');
}

export function isValidEmail(email: string): boolean {

src/services/apiClient.ts

import axios, { AxiosError, InternalAxiosRequestConfig } from 'axios';
import { API_CONFIG } from './apiConfig';

const TOKEN_STORAGE_KEY = 'codzskill_jwt_token';

export const tokenStorage = {
  get: (): string | null => localStorage.getItem(TOKEN_STORAGE_KEY),
  set: (token: string) => localStorage.setItem(TOKEN_STORAGE_KEY, token),
  clear: () => localStorage.removeItem(TOKEN_STORAGE_KEY),
};

export interface ApiError {
  status?: number;
  message: string;
  details?: Record<string, string[]>;
}

const apiClient = axios.create({
  baseURL: API_CONFIG.baseUrl,
  timeout: 15000,
  headers: {
    'Content-Type': 'application/json',
  },
});

apiClient.interceptors.request.use(
  (config: InternalAxiosRequestConfig) => {
    const token = tokenStorage.get();
    if (token && config.headers) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => Promise.reject(error)
);

apiClient.interceptors.response.use(
  (response) => response,
  (error: AxiosError<{ title?: string; message?: string; errors?: Record<string, string[]> }>) => {
    if (error.response?.status === 401) {
      tokenStorage.clear();
    }

    const normalizedError: ApiError = {
      status: error.response?.status,
      message:
        error.response?.data?.message ||
        error.response?.data?.title ||
        error.message ||
        'An unexpected network error occurred.',
      details: error.response?.data?.errors,
    };

    return Promise.reject(normalizedError);
  }
);

export default apiClient;

import React, { createContext, useContext, useEffect, useState } from 'react';
import { AuthResponse, LoginRequest, RegisterRequest, User } from '../types';
import { authService } from '../services/authService';
import { tokenStorage } from '../services/apiClient';
import { PLATFORM_MEDIA } from '../assets';

interface AuthContextValue {
  user: User | null;
  isAuthenticated: boolean;
  login: (payload: LoginRequest) => Promise<AuthResponse>;
  register: (payload: RegisterRequest) => Promise<AuthResponse>;
  switchDemoRole: (role: 'Learner' | 'Trainer') => void;
  logout: () => void;
}

const AuthContext = createContext<AuthContextValue | undefined>(undefined);
const USER_STORAGE_KEY = 'codzskill_user';

const DEFAULT_DEMO_LEARNER: User = {
  id: 'usr-learner-1',
  fullName: 'Priya Sharma',
  email: 'priya.sharma@codzskill.dev',
  role: 'Learner',
  avatarUrl: PLATFORM_MEDIA.priyaAvatar,
  headline: 'ASP.NET & React Learner',
};

export const AuthProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [user, setUser] = useState<User | null>(() => {
    const saved = localStorage.getItem(USER_STORAGE_KEY);
    if (saved) {
      try {
        return JSON.parse(saved) as User;
      } catch {
        return DEFAULT_DEMO_LEARNER;
      }
    }
    return null;
  });

  useEffect(() => {
    if (user) {
      localStorage.setItem(USER_STORAGE_KEY, JSON.stringify(user));
    } else {
      localStorage.removeItem(USER_STORAGE_KEY);
    }
  }, [user]);

  const login = async (payload: LoginRequest) => {
    const res = await authService.login(payload);
    setUser(res.user);
    return res;
  };

  const register = async (payload: RegisterRequest) => {
    const res = await authService.register(payload);
    setUser(res.user);
    return res;
  };

  const switchDemoRole = (role: 'Learner' | 'Trainer') => {
    tokenStorage.set('mock.jwt.token.codzskill.labs');
    if (role === 'Trainer') {
      setUser({
        id: 'usr-trainer-1',
        fullName: 'Rahul Verma',
        email: 'rahul.verma@codzskill.dev',
        role: 'Trainer',
        avatarUrl: PLATFORM_MEDIA.rahulAvatar,
        headline: 'Principal .NET Architect',
      });
    } else {
      setUser(DEFAULT_DEMO_LEARNER);
    }
  };

  const logout = () => {
    authService.logout();
    setUser(null);
  };

  return (
    <AuthContext.Provider
      value={{
        user,
        isAuthenticated: Boolean(user),
        login,
        register,
        switchDemoRole,
        logout,
      }}
    >
      {children}
    </AuthContext.Provider>
  );
};

export function useAuth(): AuthContextValue {
  const ctx = useContext(AuthContext);
  if (!ctx) {
    throw new Error('useAuth must be used inside AuthProvider');
  }
  return ctx;
}

src/hooks/useAsync.ts

import { useCallback, useEffect, useState } from 'react';
import { ApiError } from '../services/apiClient';

export function useAsync<T>(asyncFn: () => Promise<T>, immediate = true) {
  const [data, setData] = useState<T | null>(null);
  const [loading, setLoading] = useState<boolean>(immediate);
  const [error, setError] = useState<string | null>(null);

  const execute = useCallback(async () => {
    setLoading(true);
    setError(null);
    try {
      const result = await asyncFn();
      setData(result);
      return result;
    } catch (err) {
      const apiErr = err as ApiError;
      setError(apiErr?.message || 'Failed to load data.');
      return null;
    } finally {
      setLoading(false);
    }
  }, [asyncFn]);

  useEffect(() => {
    if (immediate) {
      void execute();
    }
  }, [execute, immediate]);

  return { data, setData, loading, error, refetch: execute };
}

import React, { useCallback, useEffect, useMemo, useState } from 'react';
import { useSearchParams } from 'react-router-dom';
import { Search, SlidersHorizontal } from 'lucide-react';
import { CourseCard } from '../components/common/CourseCard';
import { useAsync } from '../hooks/useAsync';
import { courseService } from '../services/courseService';
import { CourseCategory } from '../types';
import { cn } from '../utils/formatters';

const CATEGORIES: CourseCategory[] = [
  'All',
  'Development',
  'AI & Data',
  'Cloud & DevOps',
  'Business',
  'Design',
];

export const ExploreCoursesPage: React.FC = () => {
  const [searchParams, setSearchParams] = useSearchParams();
  const initialQuery = searchParams.get('q') || '';
  const initialCategory = (searchParams.get('category') as CourseCategory) || 'All';

  const [query, setQuery] = useState(initialQuery);
  const [activeCategory, setActiveCategory] = useState<CourseCategory>(
    CATEGORIES.includes(initialCategory) ? initialCategory : 'All'
  );
  const [showFilters, setShowFilters] = useState(false);
  const [maxPrice, setMaxPrice] = useState<number>(20000);

  const fetchCourses = useCallback(() => courseService.getCourses(), []);
  const { data: courses, loading, error } = useAsync(fetchCourses);

  useEffect(() => {
    const q = searchParams.get('q');
    const cat = searchParams.get('category') as CourseCategory | null;
    if (q !== null) setQuery(q);
    if (cat && CATEGORIES.includes(cat)) setActiveCategory(cat);
  }, [searchParams]);

  const filteredCourses = useMemo(() => {
    if (!courses) return [];
    return courses.filter((c) => {
      const matchesCat = activeCategory === 'All' || c.category === activeCategory;
      const matchesQuery =
        !query.trim() ||
        c.title.toLowerCase().includes(query.toLowerCase()) ||
        c.subtitle.toLowerCase().includes(query.toLowerCase()) ||
        c.techTag.toLowerCase().includes(query.toLowerCase());
      const matchesPrice = c.priceInr <= maxPrice;
      return matchesCat && matchesQuery && matchesPrice;
    });
  }, [courses, activeCategory, query, maxPrice]);

  const handleCategoryChange = (cat: CourseCategory) => {
    setActiveCategory(cat);
    const next = new URLSearchParams(searchParams);
    if (cat === 'All') next.delete('category');
    else next.set('category', cat);
    setSearchParams(next);
  };

  return (
    <div className="bg-slate-50 py-10">
      <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8 space-y-6">
        {/* Header + Search */}
        <div className="flex flex-col gap-4 md:flex-row md:items-center md:justify-between">
          <div>
            <h1 className="text-2xl sm:text-3xl font-bold text-slate-900">
              Explore Courses
            </h1>
            <p className="mt-1 text-sm text-slate-600">
              Learn in-demand skills with industry experts.
            </p>
          </div>

          <div className="flex items-center gap-2.5">
            <div className="relative flex-1 md:w-80">
              <Search className="absolute left-3.5 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" />
              <input
                type="text"
                value={query}
                onChange={(e) => setQuery(e.target.value)}
                placeholder="Search courses (e.g., .NET, React, AI, DevOps)..."
                className="w-full rounded-xl border border-slate-300 bg-white py-2.5 pl-10 pr-4 text-sm text-slate-900 focus:border-violet-500 focus:outline-none"
              />
            </div>
            <button
              onClick={() => setShowFilters((s) => !s)}
              className="inline-flex items-center gap-2 rounded-xl border border-slate-300 bg-white px-4 py-2.5 text-sm font-medium text-slate-700 hover:bg-slate-50"
            >
              <SlidersHorizontal className="h-4 w-4" /> Filters
            </button>
          </div>
        </div>

        {/* Optional Price Filter Drawer */}
        {showFilters && (
          <div className="rounded-2xl border border-slate-200 bg-white p-4 flex flex-wrap items-center justify-between gap-4">
            <div className="flex items-center gap-4">
              <label className="text-sm font-medium text-slate-700">
                Max Price: <span className="font-bold text-violet-600">₹{maxPrice.toLocaleString('en-IN')}</span>
              </label>
              <input
                type="range"
                min={10000}
                max={25000}
                step={1000}
                value={maxPrice}
                onChange={(e) => setMaxPrice(Number(e.target.value))}
                className="accent-violet-600"
              />
            </div>

        )}

        {/* Category Pills */}
        <div className="flex flex-wrap items-center gap-2">
          {CATEGORIES.map((cat) => (
            <button
              key={cat}

    e.preventDefault();
    if (!minLength(companyName, 2) || !minLength(contactName, 2)) {
      setError('Please enter your company name and full name.');
      return;
    }
    if (!isValidEmail(workEmail)) {
      setError('Please enter a valid work email address.');
      return;
    }

    setError('');
    setSubmitting(true);
    try {
      const res = await corporateService.submitProposal({
        companyName,
        contactName,
        workEmail,
        teamSize,
        focusArea,
        deliveryMode,
        notes,
      });
      setSuccessMessage(`${res.message} (Reference: ${res.referenceId})`);
      setCompanyName('');
      setContactName('');
      setWorkEmail('');
      setNotes('');
    } catch {
      setError('Failed to submit proposal request. Please try again.');
    } finally {
      setSubmitting(false);
    }
  };

  return (
    <div className="bg-slate-50">
      {/* Hero Banner matching screenshot */}
      <section className="relative overflow-hidden border-b border-slate-200 bg-gradient-to-br from-white via-violet-50/50 to-indigo-100/60 py-14">
        <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
          <div className="grid items-center gap-10 lg:grid-cols-12">
            <div className="lg:col-span-7 space-y-6">
              <h1 className="text-3xl sm:text-5xl font-extrabold tracking-tight text-slate-900">
                Corporate Training
              </h1>
              <p className="text-base sm:text-lg text-slate-600 max-w-xl">
                Upskill your teams with customized, industry-focused training programs.
              </p>

              <div className="grid grid-cols-2 gap-4 sm:grid-cols-4 pt-2">
                {pillars.map((p) => (
                  <div
                    key={p.title}
                    className="rounded-2xl border border-slate-200/80 bg-white p-4 shadow-sm"
                  >
                    <div className="flex h-9 w-9 items-center justify-center rounded-xl bg-violet-50">
                      {p.icon}
                    </div>
                    <h3 className="mt-3 text-sm font-bold text-slate-900">{p.title}</h3>
                    <p className="mt-1 text-xs text-slate-500">{p.desc}</p>
                  </div>
                ))}
              </div>

              <div className="flex flex-wrap gap-3 pt-2">
                <Button size="lg" onClick={scrollToForm}>
                  Request a Proposal <ArrowRight className="h-4 w-4" />
                </Button>
                <Button size="lg" variant="outline" onClick={scrollToForm}>
                  Talk to an Expert
                </Button>
              </div>
            </div>

            <div className="lg:col-span-5">
              <div className="overflow-hidden rounded-3xl border border-violet-200 bg-white p-2 shadow-xl">
                <img
                  src={PLATFORM_MEDIA.corporateTeam}
                  alt="Corporate team training session"
                  className="h-80 w-full rounded-2xl object-cover"
                />
              </div>
            </div>
          </div>
        </div>
      </section>

      {/* Business Training Programs */}
      <section className="mx-auto max-w-7xl px-4 py-14 sm:px-6 lg:px-8">
        <h2 className="text-2xl font-bold text-slate-900">
          Popular Enterprise Training Tracks
        </h2>
        <div className="mt-6 grid gap-6 md:grid-cols-3">
          {tracks.map((track) => (
            <Card key={track.title} className="p-6 flex flex-col justify-between">
              <div>
                <span className="rounded-full bg-violet-50 px-3 py-1 text-xs font-semibold text-violet-700">
                  {track.duration}
                </span>
                <h3 className="mt-3 text-lg font-bold text-slate-900">{track.title}</h3>
                <p className="mt-2 text-sm text-slate-600">{track.description}</p>
              </div>
              <button
                onClick={() => {
                  setFocusArea(track.title);
                  scrollToForm();
                }}
                className="mt-5 text-left text-xs font-semibold text-violet-600 hover:underline"
              >
                Select this track &rarr;
              </button>
            </Card>
          ))}
        </div>
      </section>

      {/* Request Proposal Form */}
      <section ref={formRef} className="mx-auto max-w-4xl px-4 pb-16 sm:px-6 lg:px-8">
        <Card className="p-6 sm:p-8">
          <h2 className="text-2xl font-bold text-slate-900">
            Request a Corporate Training Proposal
          </h2>
          <p className="mt-1 text-sm text-slate-500">
            Share your team goals and our curriculum architects will craft a custom plan.
          </p>

          {successMessage && (
            <div className="mt-5 flex items-start gap-2.5 rounded-xl border border-emerald-200 bg-emerald-50 p-4 text-sm text-emerald-800">
              <CheckCircle2 className="mt-0.5 h-5 w-5 text-emerald-600 shrink-0" />
              <span>{successMessage}</span>
            </div>
          )}

          <form onSubmit={handleSubmit} className="mt-6 grid gap-4 sm:grid-cols-2">
            <div>
              <label className="block text-sm font-medium text-slate-700">
                Company Name *
              </label>
              <input
                type="text"
                value={companyName}
                onChange={(e) => setCompanyName(e.target.value)}
                placeholder="Acme Technologies"
                className="mt-1 w-full rounded-xl border border-slate-300 px-3.5 py-2.5 text-sm focus:border-violet-500 focus:outline-none"
              />
            </div>
            <div>
              <label className="block text-sm font-medium text-slate-700">
                Your Name *
              </label>
              <input
                type="text"
                value={contactName}
                onChange={(e) => setContactName(e.target.value)}
                placeholder="Engineering Manager"
                className="mt-1 w-full rounded-xl border border-slate-300 px-3.5 py-2.5 text-sm focus:border-violet-500 focus:outline-none"
              />
            </div>

            <div>
              <label className="block text-sm font-medium text-slate-700">
                Work Email *
              </label>
              <input
                type="email"
                value={workEmail}
                onChange={(e) => setWorkEmail(e.target.value)}
                placeholder="you@company.com"
                className="mt-1 w-full rounded-xl border border-slate-300 px-3.5 py-2.5 text-sm focus:border-violet-500 focus:outline-none"
              />
            </div>
            <div>
              <label className="block text-sm font-medium text-slate-700">
                Team Size
              </label>
              <select
                value={teamSize}
                onChange={(e) => setTeamSize(e.target.value)}
                className="mt-1 w-full rounded-xl border border-slate-300 px-3.5 py-2.5 text-sm focus:border-violet-500 focus:outline-none"
              >
                <option value="10-25">10–25 Engineers</option>
                <option value="25-50">25–50 Engineers</option>
                <option value="50-150">50–150 Engineers</option>
                <option value="150+">150+ Engineers</option>
              </select>
            </div>

            <div>
              <label className="block text-sm font-medium text-slate-700">
                Primary Technology Track
              </label>
              <input
                type="text"
                value={focusArea}
                onChange={(e) => setFocusArea(e.target.value)}
                className="mt-1 w-full rounded-xl border border-slate-300 px-3.5 py-2.5 text-sm focus:border-violet-500 focus:outline-none"
              />
            </div>
            <div>
              <label className="block text-sm font-medium text-slate-700">
                Delivery Mode
              </label>
              <select
                value={deliveryMode}
                onChange={(e) =>
                  setDeliveryMode(e.target.value as 'Online' | 'On-site' | 'Hybrid')
                }
                className="mt-1 w-full rounded-xl border border-slate-300 px-3.5 py-2.5 text-sm focus:border-violet-500 focus:outline-none"
              >
                <option value="Online">Online Live</option>
                <option value="On-site">On-site Workshop</option>
                <option value="Hybrid">Hybrid</option>
              </select>
            </div>

            <div className="sm:col-span-2">
              <label className="block text-sm font-medium text-slate-700">
                Training Goals &amp; Timeline
              </label>
              <textarea
                rows={3}
                value={notes}
                onChange={(e) => setNotes(e.target.value)}
                placeholder="Tell us about your current tech stack and upskilling timeline..."
                className="mt-1 w-full rounded-xl border border-slate-300 px-3.5 py-2.5 text-sm focus:border-violet-500 focus:outline-none"
              />
            </div>

            {error && (
              <div className="sm:col-span-2 rounded-xl bg-rose-50 px-3 py-2 text-xs font-medium text-rose-700">
                {error}
              </div>
            )}

            <div className="sm:col-span-2">
              <Button type="submit" size="lg" loading={submitting}>
                Submit Proposal Request <ArrowRight className="h-4 w-4" />
              </Button>
            </div>
          </form>
        </Card>
      </section>
    </div>
  );
};

18. Running & Connecting to ASP.NET Core 10 Web API

1.  Install and start locally:
    npm install
    cp .env.example .env
    npm run dev
2.  Switch from Mock Data to your ASP.NET Core 10 API:
      - Set VITE_USE_MOCK_DATA=false and
        VITE_API_BASE_URL=https://localhost:7001/api in .env.
      - Update route paths in src/services/apiConfig.ts (or via VITE_EP_*
        environment variables) to match your ASP.NET Core controllers.
      - src/services/apiClient.ts automatically attaches Authorization: Bearer
        <token> on every request and handles 401 token cleanup cleanly.
